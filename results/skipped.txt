#!/bin/bash
set -uo pipefail

SOURCE_NS="test-pods"
TARGET_NS="default"

SOURCE_POD=$(kubectl get pods -n $SOURCE_NS -l app=metrics-collector -o jsonpath='{.items[0].metadata.name}')
if [ -z "$SOURCE_POD" ]; then echo "SKIP: source pod not found in ns $SOURCE_NS"; exit 0; fi

kubectl get svc any -n $TARGET_NS >/dev/null 2>&1 || { echo "SKIP: target service not found in ns $TARGET_NS"; exit 0; }
SVC_IP=$(kubectl get svc any -n $TARGET_NS -o jsonpath='{.spec.clusterIP}')
SVC_PORT=$(kubectl get svc any -n $TARGET_NS -o jsonpath='{.spec.ports[0].port}')

echo "=== TEST START ==="
RESULT=0
CMD_OUTPUT=""
EXIT_CODE=""
for i in 1 2 3 4 5; do
  CMD_OUTPUT=$(kubectl exec -n $SOURCE_NS $SOURCE_POD -- nc -z -w 3 $SVC_IP $SVC_PORT 2>&1)
  EXIT_CODE=$?
  SUCCESS_COND=$( [ $EXIT_CODE -eq 0 ] && echo 1 || echo 0 )
  if [ $SUCCESS_COND -eq 1 ]; then
    echo "SUCCESS: Connection behaved as expected."
    RESULT=1
    break
  fi
  echo "Attempt $i failed. Tool output: $CMD_OUTPUT"
  sleep 2
done
if [ $RESULT -eq 0 ]; then
  echo "FAIL: Connection did not behave as expected."
  echo "Last tool output: $CMD_OUTPUT"
  echo "Last exit code: $EXIT_CODE"
  exit 1
fi
exit 0
